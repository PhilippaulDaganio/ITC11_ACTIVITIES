#include <iostream>
using namespace std;

int main() {
    // Declare variables
    double num1, num2, result;
    char operation;
    char choice;

    do {
        // Display menu
        cout << "\n==============================" << endl;
        cout << "    SIMPLE CALCULATOR         " << endl;
        cout << "==============================" << endl;
        cout << "  [+] Addition" << endl;
        cout << "  [-] Subtraction" << endl;
        cout << "  [*] Multiplication" << endl;
        cout << "  [/] Division" << endl;
        cout << "==============================" << endl;

        // Get user input
        cout << "Enter first number: ";
        cin >> num1;

        cout << "Enter operation (+, -, *, /): ";
        cin >> operation;

        cout << "Enter second number: ";
        cin >> num2;

        // Perform calculation using switch
        switch (operation) {
            case '+':
                result = num1 + num2;
                cout << "\nResult: " << num1 << " + " << num2 << " = " << result << endl;
                break;

            case '-':
                result = num1 - num2;
                cout << "\nResult: " << num1 << " - " << num2 << " = " << result << endl;
                break;

            case '*':
                result = num1 * num2;
                cout << "\nResult: " << num1 << " * " << num2 << " = " << result << endl;
                break;

            case '/':
                if (num2 != 0) {
                    result = num1 / num2;
                    cout << "\nResult: " << num1 << " / " << num2 << " = " << result << endl;
                } else {
                    cout << "\nError: Division by zero is not allowed!" << endl;
                }
                break;

            default:
                cout << "\nError: Invalid operation!" << endl;
        }

        // Ask if user wants to continue
        cout << "\nDo you want to calculate again? (y/n): ";
        cin >> choice;

    } while (choice == 'y' || choice == 'Y');

    cout << "\nThank you for using the calculator. Goodbye!" << endl;

    return 0;
}
